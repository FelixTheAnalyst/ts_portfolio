Overview

This section covers models for forecasting and for structure that changes over time. Exponential smoothing is the simplest state-space model. Regression adds explanatory terms such as harmonics, indicator variables and count-data models. The Kalman filter treats a quantity like a level or a regression coefficient as an unobserved state that moves over time. Hidden Markov models do the same for a discrete state, so the series switches between regimes.

I used the regression material in Project 3 (a negative-binomial model for withdrawals) and the state-space material in Project 4 (a time-varying beta). The tutorials here compare five forecasting methods on Eskom’s EAF, then build a hidden Markov model from first principles and test whether its regime lengths behave as the model assumes.

Tutorial 2.1: Forecasting EAF by smoothing, regression and ARIMA

Aim. Compare five forecasting methods on the same series by repeatedly forecasting a quarter ahead from many starting points, then forecast 26 weeks ahead with the best one.

The methods are a naive forecast (repeat the last value), simple exponential smoothing, damped-trend exponential smoothing, an automatically selected ARIMA, and harmonic regression: a linear trend plus two annual sine-cosine pairs, with AR(1) errors.

nf <- function(tr) ts(as.numeric(tr), frequency = 1)   # drop the weekly frequency for non-seasonal methods
fc_fun <- list(
  Naive = function(tr, h) forecast::rwf(nf(tr), h = h),
  `Simple smoothing` = function(tr, h) forecast::forecast(forecast::ets(nf(tr), model = "ANN"), h = h),
  `Damped trend` = function(tr, h) forecast::forecast(forecast::ets(nf(tr), model = "AAN", damped = TRUE), h = h),
  ARIMA = function(tr, h) forecast::forecast(forecast::auto.arima(nf(tr), seasonal = FALSE), h = h),
  `Harmonic regression` = function(tr, h) {
    nt <- length(tr)
    xr  <- cbind(trend = seq_len(nt), forecast::fourier(tr, K = 2))                          # trend + 2 harmonics
    fit <- forecast::Arima(tr, order = c(1, 0, 0), xreg = xr)                                # AR(1) errors
    xf  <- cbind(trend = nt + seq_len(h), forecast::fourier(tr, K = 2, h = h))               # future regressors
    forecast::forecast(fit, h = h, xreg = xf)
  })

Implementation 2.2: A hidden Markov model from first principles

Aim. Describe EAF as a Gaussian hidden Markov model with m regimes (for example normal and stressed operation), estimate it by the Baum-Welch (EM) algorithm that I code myself, decode the most likely regime path with the Viterbi algorithm, and choose the number of regimes by BIC.

With hidden state S_t\in\{1,\dots,m\}, transition matrix A and X_t\mid S_t=i\sim N(\mu_i,\sigma_i^2), the likelihood is computed by the forward recursion. I scale each step, because unscaled probabilities underflow over 232 weeks:

\hat\alpha_t(j)=\frac{\big[\sum_i\hat\alpha_{t-1}(i)a_{ij}\big]\,b_j(x_t)}{c_t},\qquad \log L=\sum_t\log c_t.

The backward pass and the EM updates follow from the posterior state probabilities \gamma_t(i) and the expected transitions \xi_t(i,j).

# State-conditional densities: T x m matrix
hmm_dens <- function(x, mu, sg) outer(x, seq_along(mu), function(xx, j) dnorm(xx, mu[j], sg[j]))

# Scaled forward algorithm: filtered state probabilities and the log-likelihood
hmm_forward <- function(x, mu, sg, A, delta) {
  nT <- length(x); m <- length(mu); B <- hmm_dens(x, mu, sg)
  al <- matrix(0, nT, m); cc <- numeric(nT)
  a <- delta * B[1, ]; cc[1] <- sum(a); al[1, ] <- a / cc[1]
  for (t in 2:nT) { a <- drop(al[t - 1, ] %*% A) * B[t, ]; cc[t] <- sum(a); al[t, ] <- a / cc[t] }
  list(loglik = sum(log(cc)), alpha = al, c = cc, B = B)
}

# Scaled backward algorithm, using the forward scaling constants
hmm_backward <- function(A, B, cc) {
  nT <- nrow(B); be <- matrix(1, nT, ncol(B))
  for (t in (nT - 1):1) be[t, ] <- drop(A %*% (B[t + 1, ] * be[t + 1, ])) / cc[t + 1]
  be
}

# Baum-Welch: EM until the log-likelihood stops improving
baum_welch <- function(x, mu, sg, A, delta, maxit = 500, tol = 1e-9) {
  m <- length(mu); nT <- length(x); ll_old <- -Inf; lls <- numeric(0)
  for (it in seq_len(maxit)) {
    fw <- hmm_forward(x, mu, sg, A, delta); be <- hmm_backward(A, fw$B, fw$c)
    g <- fw$alpha * be                                               # P(state at t | all data)
    xi <- matrix(0, m, m)                                            # expected transition counts
    for (t in 1:(nT - 1)) xi <- xi + outer(fw$alpha[t, ], fw$B[t + 1, ] * be[t + 1, ]) * A / fw$c[t + 1]
    lls <- c(lls, fw$loglik)
    delta <- g[1, ]
    A  <- xi / colSums(g[-nT, , drop = FALSE])                       # each row divided by its expected visits
    w  <- colSums(g); mu <- colSums(g * x) / w                       # weighted means
    sg <- pmax(sqrt(colSums(g * (x - rep(mu, each = nT))^2) / w), 1e-3 * sd(x))   # weighted sds, floored
    if (abs(fw$loglik - ll_old) < tol * (abs(fw$loglik) + tol)) break
    ll_old <- fw$loglik
  }
  list(mu = mu, sg = sg, A = A, delta = delta, loglik = hmm_forward(x, mu, sg, A, delta)$loglik, lls = lls, iter = it)
}

# Several starts (quantile-based means, sticky transitions); keep the best; order states by mean
fit_hmm <- function(x, m, nstart = 10, seed = 1) {
  set.seed(seed); best <- NULL
  for (s in seq_len(nstart)) {
    mu0 <- as.numeric(quantile(x, seq(0.5 / m, 1 - 0.5 / m, length.out = m)))
    if (s > 1) mu0 <- sort(mu0 + rnorm(m, 0, 0.1 * sd(x)))           # jitter after the first start
    A0 <- matrix(0.05 / (m - 1), m, m); diag(A0) <- 0.95             # rows sum to one
    f <- baum_welch(x, mu0, rep(sd(x) / m, m), A0, rep(1 / m, m))
    if (is.null(best) || f$loglik > best$loglik) best <- f
  }
  o <- order(best$mu)
  best$mu <- best$mu[o]; best$sg <- best$sg[o]; best$A <- best$A[o, o, drop = FALSE]; best$delta <- best$delta[o]
  best
}

# Viterbi: most likely state path, computed on the log scale
viterbi <- function(x, mu, sg, A, delta) {
  nT <- length(x); m <- length(mu)
  lB <- log(hmm_dens(x, mu, sg) + 1e-300); lA <- log(A + 1e-300)
  v <- matrix(0, nT, m); bp <- matrix(0L, nT, m); v[1, ] <- log(delta + 1e-300) + lB[1, ]
  for (t in 2:nT) for (j in 1:m) { s <- v[t - 1, ] + lA[, j]; bp[t, j] <- which.max(s); v[t, j] <- max(s) + lB[t, j] }
  st <- integer(nT); st[nT] <- which.max(v[nT, ])
  for (t in (nT - 1):1) st[t] <- bp[t + 1, st[t + 1]]
  st
}

Limitations. EM finds local optima, so I use several starts. The model treats the Gaussian state-conditional distribution as fixed within a regime, but EAF has a trend and an annual cycle, so regimes partly absorb them. A hidden Markov model is therefore a description of regimes, not a causal account, and the number of regimes depends on the criterion used.

Beyond the notes 2.3: Do the regimes last as long as the model says?

Idea. A hidden Markov model implies that the time spent in a regime is geometric, with mean 1/(1-a_{ii}). That is a strong assumption, and the decoded path lets me test it by comparing run lengths with the model’s expectation.

I would still test the duration distribution directly with a longer series.

Reflection

  • Tutorials: “harmonic regression won with an RMSE of 3.08, 21% below the naive forecast, while ARIMA and the smoothing methods only reached 0.96 to 0.97 of the naive RMSE; fifteen origins is a small number, so I trust the ranking of the winner against the rest more than the order among close competitors”

  • HMM: “BIC chose four regimes (1190.7 against 1223.4 for three), with means of 52.8, 58.4, 63.9 and 69.6. Their roughly even spacing, together with the steady fall in EAF over the window, makes me think the regimes partly absorb the trend as well as describing genuine operating states, so I would not interpret them as separate physical regimes without a longer series.”

# Tutorial 2.1: Forecasting EAF by smoothing, regression and ARIMA

**Aim.** Compare five forecasting methods on the same series by repeatedly forecasting a quarter ahead from many starting points, then forecast 26 weeks ahead with the best one.

The methods are a naive forecast (repeat the last value), simple exponential smoothing, damped-trend exponential smoothing, an automatically selected ARIMA, and **harmonic regression**: a linear trend plus two annual sine-cosine pairs, with AR(1) errors.

# Implementation 2.2: A hidden Markov model from first principles

**Aim.** Describe EAF as a Gaussian hidden Markov model with $m$ regimes (for example normal and stressed operation), estimate it by the Baum-Welch (EM) algorithm that I code myself, decode the most likely regime path with the Viterbi algorithm, and choose the number of regimes by BIC.

With hidden state $S_t\in\{1,\dots,m\}$, transition matrix $A$ and $X_t\mid S_t=i\sim N(\mu_i,\sigma_i^2)$, the likelihood is computed by the forward recursion. I scale each step, because unscaled probabilities underflow over 232 weeks:

$$\hat\alpha_t(j)=\frac{\big[\sum_i\hat\alpha_{t-1}(i)a_{ij}\big]\,b_j(x_t)}{c_t},\qquad \log L=\sum_t\log c_t.$$

The backward pass and the EM updates follow from the posterior state probabilities $\gamma_t(i)$ and the expected transitions $\xi_t(i,j)$.

```{r}
#| label: hmm-functions
#| code-fold: false
#| code-summary: "Forward, backward, Baum-Welch EM and Viterbi for a Gaussian HMM"

# State-conditional densities: T x m matrix
hmm_dens <- function(x, mu, sg) outer(x, seq_along(mu), function(xx, j) dnorm(xx, mu[j], sg[j]))

# Scaled forward algorithm: filtered state probabilities and the log-likelihood
hmm_forward <- function(x, mu, sg, A, delta) {
  nT <- length(x); m <- length(mu); B <- hmm_dens(x, mu, sg)
  al <- matrix(0, nT, m); cc <- numeric(nT)
  a <- delta * B[1, ]; cc[1] <- sum(a); al[1, ] <- a / cc[1]
  for (t in 2:nT) { a <- drop(al[t - 1, ] %*% A) * B[t, ]; cc[t] <- sum(a); al[t, ] <- a / cc[t] }
  list(loglik = sum(log(cc)), alpha = al, c = cc, B = B)
}

# Scaled backward algorithm, using the forward scaling constants
hmm_backward <- function(A, B, cc) {
  nT <- nrow(B); be <- matrix(1, nT, ncol(B))
  for (t in (nT - 1):1) be[t, ] <- drop(A %*% (B[t + 1, ] * be[t + 1, ])) / cc[t + 1]
  be
}

# Baum-Welch: EM until the log-likelihood stops improving
baum_welch <- function(x, mu, sg, A, delta, maxit = 500, tol = 1e-9) {
  m <- length(mu); nT <- length(x); ll_old <- -Inf; lls <- numeric(0)
  for (it in seq_len(maxit)) {
    fw <- hmm_forward(x, mu, sg, A, delta); be <- hmm_backward(A, fw$B, fw$c)
    g <- fw$alpha * be                                               # P(state at t | all data)
    xi <- matrix(0, m, m)                                            # expected transition counts
    for (t in 1:(nT - 1)) xi <- xi + outer(fw$alpha[t, ], fw$B[t + 1, ] * be[t + 1, ]) * A / fw$c[t + 1]
    lls <- c(lls, fw$loglik)
    delta <- g[1, ]
    A  <- xi / colSums(g[-nT, , drop = FALSE])                       # each row divided by its expected visits
    w  <- colSums(g); mu <- colSums(g * x) / w                       # weighted means
    sg <- pmax(sqrt(colSums(g * (x - rep(mu, each = nT))^2) / w), 1e-3 * sd(x))   # weighted sds, floored
    if (abs(fw$loglik - ll_old) < tol * (abs(fw$loglik) + tol)) break
    ll_old <- fw$loglik
  }
  list(mu = mu, sg = sg, A = A, delta = delta, loglik = hmm_forward(x, mu, sg, A, delta)$loglik, lls = lls, iter = it)
}

# Several starts (quantile-based means, sticky transitions); keep the best; order states by mean
fit_hmm <- function(x, m, nstart = 10, seed = 1) {
  set.seed(seed); best <- NULL
  for (s in seq_len(nstart)) {
    mu0 <- as.numeric(quantile(x, seq(0.5 / m, 1 - 0.5 / m, length.out = m)))
    if (s > 1) mu0 <- sort(mu0 + rnorm(m, 0, 0.1 * sd(x)))           # jitter after the first start
    A0 <- matrix(0.05 / (m - 1), m, m); diag(A0) <- 0.95             # rows sum to one
    f <- baum_welch(x, mu0, rep(sd(x) / m, m), A0, rep(1 / m, m))
    if (is.null(best) || f$loglik > best$loglik) best <- f
  }
  o <- order(best$mu)
  best$mu <- best$mu[o]; best$sg <- best$sg[o]; best$A <- best$A[o, o, drop = FALSE]; best$delta <- best$delta[o]
  best
}

# Viterbi: most likely state path, computed on the log scale
viterbi <- function(x, mu, sg, A, delta) {
  nT <- length(x); m <- length(mu)
  lB <- log(hmm_dens(x, mu, sg) + 1e-300); lA <- log(A + 1e-300)
  v <- matrix(0, nT, m); bp <- matrix(0L, nT, m); v[1, ] <- log(delta + 1e-300) + lB[1, ]
  for (t in 2:nT) for (j in 1:m) { s <- v[t - 1, ] + lA[, j]; bp[t, j] <- which.max(s); v[t, j] <- max(s) + lB[t, j] }
  st <- integer(nT); st[nT] <- which.max(v[nT, ])
  for (t in (nT - 1):1) st[t] <- bp[t + 1, st[t + 1]]
  st
}
```

## Checking the code

```{r}
#| label: hmm-validate

# Check 1: the forward log-likelihood must equal a brute-force sum over every state path (T = 6, m = 2)
set.seed(1); xs <- rnorm(6, 0, 2)
mu_t <- c(0, 3); sg_t <- c(1, 1.5); A_t <- matrix(c(.9, .1, .2, .8), 2, byrow = TRUE); d_t <- c(.5, .5)
brute <- sum(apply(expand.grid(rep(list(1:2), 6)), 1, function(p) {
  pr <- d_t[p[1]] * dnorm(xs[1], mu_t[p[1]], sg_t[p[1]])
  for (t in 2:6) pr <- pr * A_t[p[t - 1], p[t]] * dnorm(xs[t], mu_t[p[t]], sg_t[p[t]])
  pr
}))
diff_fw <- abs(hmm_forward(xs, mu_t, sg_t, A_t, d_t)$loglik - log(brute))

# Check 2: simulate a 3-state HMM with known parameters, then recover them
sim_hmm <- function(nT, mu, sg, A, seed = 2026) {
  set.seed(seed); m <- length(mu); s <- integer(nT); s[1] <- 2L
  for (t in 2:nT) s[t] <- sample.int(m, 1, prob = A[s[t - 1], ])
  list(s = s, x = rnorm(nT, mu[s], sg[s]))
}
mu_s <- c(54, 62, 70); sg_s <- c(2.5, 2.5, 2); A_s <- matrix(c(.93, .07, 0, .04, .92, .04, 0, .08, .92), 3, byrow = TRUE)

mono <- all(diff(fs$lls) > -1e-8)                                           # EM must never decrease the likelihood
```

# Beyond the notes 2.3: Do the regimes last as long as the model says?

**Idea.** A hidden Markov model implies that the time spent in a regime is geometric, with mean $1/(1-a_{ii})$. That is a strong assumption, and the decoded path lets me test it by comparing run lengths with the model's expectation.

```{r}
#| label: tbl-runs
#| tbl-cap: "Observed regime run lengths against the geometric expectation implied by the fitted transition matrix."

rl <- rle(st)
tibble(state = rl$values, len = rl$lengths) |>
  group_by(Regime = state) |>
  summarise(Runs = n(), `Observed mean run (weeks)` = mean(len), `Longest run` = max(len), .groups = "drop") |>
  mutate(`Expected under the HMM` = 1 / (1 - diag(fb$A)[Regime])) |> kable(digits = 1)
```

Observed mean run lengths are close to the geometric expectation for regimes 2, 3 and 4 (12.6 against 11.8, 7.2 against 7.3, and 18.0 against 17.1 weeks). Regime 1 is shorter than expected (10.7 against 14.0). With only 21 runs in total, this is too few to reject the Markov assumption, so the data give no clear reason to prefer a semi-Markov model. I would still test the duration distribution directly with a longer series.
